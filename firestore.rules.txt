rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions for institutional authorization
    function isAuthenticated() {
      return request.auth != null;
    }
    
    function isAdmin() {
      return isAuthenticated() && 
        (request.auth.token.admin == true || 
         request.auth.token.role == 'admin');
    }

    function isOwner(studentId) {
      return isAuthenticated() && request.auth.uid == studentId;
    }

    // Question Bank: Public can read published items; Only verified admins can write
    match /questions/{questionId} {
      allow read: if resource.data.status == 'Published' || isAdmin();
      allow create, update, delete: if isAdmin();
    }

    // Attempt Snapshots: Students can write their own attempts and read only their own records
    match /attempts/{attemptId} {
      allow read: if isAdmin() || (isAuthenticated() && resource.data.studentId == request.auth.uid);
      allow create: if isAuthenticated() && request.resource.data.studentId == request.auth.uid;
      allow update, delete: if isAdmin();
    }

    // Student Profiles
    match /students/{studentId} {
      allow read, write: if isOwner(studentId) || isAdmin();
    }
  }
}
